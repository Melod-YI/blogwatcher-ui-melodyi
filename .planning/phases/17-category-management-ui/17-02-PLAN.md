       ```go
       func (s *Server) handleCategoriesCreate(w http.ResponseWriter, r *http.Request) {
           name := r.FormValue("name")
           if name == "" {
               http.Error(w, "Category name required", http.StatusBadRequest)
               return
           }

           category, err := s.db.CreateCategory(name)
           if err != nil {
               log.Printf("Error creating category: %v", err)
               http.Error(w, "Failed to create category", http.StatusInternalServerError)
               return
           }

           // 返回分类 display 行，触发 categoryListUpdated
           blogCount := 0 // 新分类没有 blog
           data := map[string]interface{}{
               "Category":  category,
               "BlogCount": blogCount,
           }
           w.Header().Set("HX-Trigger", "categoryListUpdated")
           s.renderTemplate(w, "category-item.gohtml", data)
       }
       ```

    3. **handleCategoryEdit (GET /categories/{id}/edit)**：
       ```go
       func (s *Server) handleCategoryEdit(w http.ResponseWriter, r *http.Request) {
           idStr := r.PathValue("id")
           id, err := strconv.ParseInt(idStr, 10, 64)
           if err != nil {
               http.Error(w, "Invalid category ID", http.StatusBadRequest)
               return
           }

           // 获取分类信息
           categories, err := s.db.ListCategoriesWithBlogCount()
           if err != nil {
               log.Printf("Error fetching categories: %v", err)
               http.Error(w, "Database error", http.StatusInternalServerError)
               return
           }

           var category *storage.CategoryWithBlogCount
           for _, c := range categories {
               if c.ID == id {
                   category = &c
                   break
               }
           }
           if category == nil {
               http.Error(w, "Category not found", http.StatusNotFound)
               return
           }

           data := map[string]interface{}{
               "Category":  category,
               "BlogCount": category.BlogCount,
           }
           s.renderTemplate(w, "category-edit-form.gohtml", data)
       }
       ```

    4. **handleCategoryUpdate (PUT /categories/{id})**：
       ```go
       func (s *Server) handleCategoryUpdate(w http.ResponseWriter, r *http.Request) {
           idStr := r.PathValue("id")
           id, err := strconv.ParseInt(idStr, 10, 64)
           if err != nil {
               http.Error(w, "Invalid category ID", http.StatusBadRequest)
               return
           }

           name := r.FormValue("name")
           if name == "" {
               http.Error(w, "Category name required", http.StatusBadRequest)
               return
           }

           err = s.db.UpdateCategoryName(id, name)
           if err != nil {
               log.Printf("Error updating category: %v", err)
               http.Error(w, "Failed to update category", http.StatusInternalServerError)
               return
           }

           // 获取更新后的分类 + blog count
           categories, err := s.db.ListCategoriesWithBlogCount()
           if err != nil {
               log.Printf("Error fetching categories: %v", err)
               http.Error(w, "Database error", http.StatusInternalServerError)
               return
           }

           var category *storage.CategoryWithBlogCount
           for _, c := range categories {
               if c.ID == id {
                   category = &c
                   break
               }
           }

           data := map[string]interface{}{
               "Category":  category,
               "BlogCount": category.BlogCount,
           }
           w.Header().Set("HX-Trigger", "categoryListUpdated")
           s.renderTemplate(w, "category-item.gohtml", data)
       }
       ```

    5. **handleCategoryDelete (DELETE /categories/{id})**：
       ```go
       func (s *Server) handleCategoryDelete(w http.ResponseWriter, r *http.Request) {
           idStr := r.PathValue("id")
           id, err := strconv.ParseInt(idStr, 10, 64)
           if err != nil {
               http.Error(w, "Invalid category ID", http.StatusBadRequest)
               return
           }

           err = s.db.DeleteCategory(id)
           if err != nil {
               log.Printf("Error deleting category: %v", err)
               http.Error(w, "Failed to delete category", http.StatusInternalServerError)
               return
           }

           // 返回空响应（分类已删除），触发刷新
           w.Header().Set("HX-Trigger", "categoryListUpdated, blogListUpdated")
           // 返回空字符串，让 HTMX 删除元素
           w.WriteHeader(http.StatusOK)
       }
       ```

| Handler → Database | 参数化查询防止 SQL injection |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-17-05 | Tampering | handleCategoriesCreate name input | mitigate | 验证 name 不为空，使用 FormValue 获取 |
| T-17-06 | Tampering | handleCategoryUpdate name input | mitigate | 验证 name 不为空 |
| T-17-07 | Tampering | handleCategoryDelete id parameter | mitigate | 使用 PathValue + ParseInt，验证数值有效性 |
| T-17-08 | Information Disclosure | Error messages | accept | 低价值数据，错误信息不暴露敏感信息 |
| T-17-09 | Elevation of Privilege | Category operations | accept | 单用户系统，无权限控制需求 |

所有 Handler 使用参数化查询（database.go 方法），HTMX POST/PUT/DELETE 无需 CSRF（单用户本地应用）。
