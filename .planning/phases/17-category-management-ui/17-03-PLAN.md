       ```go
       func (s *Server) handleBlogUpdate(w http.ResponseWriter, r *http.Request) {
           idStr := r.PathValue("id")
           id, err := strconv.ParseInt(idStr, 10, 64)
           if err != nil {
               http.Error(w, "Invalid blog ID", http.StatusBadRequest)
               return
           }

           name := r.FormValue("name")
           if name == "" {
               http.Error(w, "Blog name required", http.StatusBadRequest)
               return
           }

           // Update blog name
           err = s.db.UpdateBlogName(id, name)
           if err != nil {
               log.Printf("Error updating blog name: %v", err)
               http.Error(w, "Failed to update blog", http.StatusInternalServerError)
               return
           }

           // NEW: Update blog category
           categoryIDStr := r.FormValue("category_id")
           var categoryID *int64
           if categoryIDStr != "" {
               catID, err := strconv.ParseInt(categoryIDStr, 10, 64)
               if err != nil {
                   log.Printf("Invalid category ID: %v", err)
                   // 不阻止更新，忽略无效分类
               } else {
                   categoryID = &catID
               }
           }

           err = s.db.UpdateBlogCategory(id, categoryID)
           if err != nil {
               log.Printf("Error updating blog category: %v", err)
               // 不阻止响应，category 更新失败不影响 name 更新成功
           }

           // Return display row with HX-Trigger for sidebar refresh
           blog, err := s.db.GetBlogByID(id)
           if err != nil || blog == nil {
               http.Error(w, "Blog not found", http.StatusNotFound)
               return
           }

           articleCount, err := s.db.GetArticleCountForBlog(id)
           if err != nil {
               articleCount = 0
           }

           data := map[string]interface{}{
               "Blog":         blog,
               "ArticleCount": articleCount,
           }
           w.Header().Set("HX-Trigger", "blogListUpdated")
           s.renderTemplate(w, "blog-display-row.gohtml", data)
       }
       ```
