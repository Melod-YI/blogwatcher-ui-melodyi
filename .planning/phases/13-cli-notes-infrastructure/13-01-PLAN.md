func (db *Database) columnExists(table, column string) bool {
    rows, err := db.conn.Query("PRAGMA table_info(" + table + ")")
    if err != nil {
        return false
    }
    defer rows.Close()
    for rows.Next() {
        var cid int
        var name, ctype string
        var notnull, pk int
        var dfltValue any
        if err := rows.Scan(&cid, &name, &ctype, &notnull, &dfltValue, &pk); err != nil {
            continue
        }
        if name == column {
            return true
        }
    }
    return false
}
```

// UpdateArticleHasNote updates the has_note field for an article.
// Returns error if the article does not exist.
func (db *Database) UpdateArticleHasNote(id int64, hasNote bool) error {
    result, err := db.conn.Exec(`UPDATE articles SET has_note = ? WHERE id = ?`, hasNote, id)
    if err != nil {
        return err
    }
    rowsAffected, err := result.RowsAffected()
    if err != nil {
        return err
    }
    if rowsAffected == 0 {
        return fmt.Errorf("article not found: %d", id)
    }
    return nil
}
```

**关键实现细节：**
- 方法签名：`func (db *Database) UpdateArticleHasNote(id int64, hasNote bool) error`

- 使用 Exec + RowsAffected 模式（与 MarkArticleRead 一致）
- 检查 rowsAffected == 0 时返回错误：`article not found: %d`
- 参数顺序：hasNote, id（与 SQL 语句 ? 顺序一致）
